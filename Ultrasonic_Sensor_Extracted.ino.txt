#define TRIG_PIN 9  // Trig pin of HC-SR04
#define ECHO_PIN 10 // Echo pin of HC-SR04
#define RELAY_PIN 8 // Relay module control pin

long duration;
int distance;

void setup() {
  pinMode(TRIG_PIN, OUTPUT); // Trig pin as output
  pinMode(ECHO_PIN, INPUT); // Echo pin as input
  pinMode(RELAY_PIN, OUTPUT); // Relay pin as output
  Serial.begin(9600); // Initialize serial communication for debugging
}

void loop() {
  // Generate a 10-microsecond pulse to trigger the sensor
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  // Measure the duration of the echo pulse
  duration = pulseIn(ECHO_PIN, HIGH);

  // Calculate distance in centimeters
  distance = duration * 0.034 / 2;

  // Print distance to Serial Monitor
  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  // Control motor via relay based on distance
  if (distance > 0 && distance <= 20) { // Object detected within 20 cm
    digitalWrite(RELAY_PIN, HIGH); // Turn on motor
    Serial.println("Motor ON");
  } else {
    // The report ends here; the remaining code is not shown.
  }
}
